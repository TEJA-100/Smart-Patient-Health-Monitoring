String urlEncode(const String &s) {
  String out = "";
  char buf[4];
  for (size_t i = 0; i < s.length(); i++) {
    char c = s[i];
    if (isalnum(c) || c == '-' || c == '_' || c == '.' || c == '~') {
      out += c;
    } else {
      snprintf(buf, sizeof(buf), "%%%02X", (unsigned char)c);
      out += buf;
    }
  }
  return out;
}

bool sendTelegramMessage(const String &text) {
  if (WiFi.status() != WL_CONNECTED) return false;

  WiFiClientSecure client;
  client.setInsecure();   // skips certificate check (fine for a school project)

  HTTPClient https;
  String url = "https://api.telegram.org/bot" + String(BOT_TOKEN) +
               "/sendMessage?chat_id=" + String(CHAT_ID) +
               "&text=" + urlEncode(text);

  if (!https.begin(client, url)) return false;
  int code = https.GET();
  https.end();

  Serial.print("[Telegram] Response code: ");
  Serial.println(code);
  return code == 200;
}

  json += "\"bpm\":" + String(smoothedBPM, 1) + ",";
  json += "\"temp\":" + String(temperatureC, 2) + ",";
  json += "\"status\":\"" + healthStatus + "\"";
  json += "}";
  server.send(200, "application/json", json);
}

// Last 60 seconds of BPM and temperature, oldest first
void handleHistory() {
  String json;
  json.reserve(700);
  int start = (histHead - histCount + HIST) % HIST;

  json = "{\"bpm\":[";
  for (int i = 0; i < histCount; i++) {
    int idx = (start + i) % HIST;
    if (i) json += ",";
    json += String(bpmHist[idx], 0);
  }
  json += "],\"temp\":[";
  for (int i = 0; i < histCount; i++) {
    int idx = (start + i) % HIST;
    if (i) json += ",";
    json += String(tempHist[idx], 1);
  }
  json += "]}";
  server.send(200, "application/json", json);
}

// ================= SIMULATED TEMPERATURE (PCB UNCHANGED) =================
// DEMONSTRATION ONLY: these are simulated values, NOT actual body temperature.
// Values vary between 70.0 and 105.0 °C for software/project testing.
float readTemperature() {
  return random(700, 1051) / 10.0;
}

void setup() {
  // Start Serial Monitor immediately
  Serial.begin(115200);
  randomSeed(esp_random()); // Seed simulated temperature variation
  delay(1500); // Allow USB-Serial connection to stabilize
  Serial.println("\n=========================================");
  Serial.println("  ESP32 HEALTH MONITOR INITIALIZING...   ");
  Serial.println("=========================================");

  pinMode(BUZZER_PIN, OUTPUT);
  analogReadResolution(12);

  // Power-on test beep
  tone(BUZZER_PIN, 2000, 80);
  delay(100);
  noTone(BUZZER_PIN);

  // Wi-Fi Connection with 10-Second Timeout
  Serial.print("Connecting to Wi-Fi: ");
  Serial.println(ssid);
  WiFi.begin(ssid, password);

  unsigned long wifiStart = millis();
  while (WiFi.status() != WL_CONNECTED && (millis() - wifiStart < 10000)) {
    delay(500);
    Serial.print(".");
  }

  if (WiFi.status() == WL_CONNECTED) {
    wifiConnected = true;
    Serial.println("\n[OK] WiFi Connected!");
    Serial.print("[INFO] Dashboard URL: http://");
    Serial.println(WiFi.localIP());

    server.on("/", handleRoot);
    server.on("/data", handleData);
    server.on("/history", handleHistory);
    server.begin();

    // Telegram startup message
    String startMsg = "✅ Patient monitor is online\n";
    startMsg += "📊 Live Dashboard, Graphs & Analysis\n";
    startMsg += "http://" + WiFi.localIP().toString() + "/";
    sendTelegramMessage(startMsg);
    lastTelegramTime = millis();
  } else {
    Serial.println("\n[WARN] WiFi connection timed out (Wrong SSID/Password or out of range).");
    Serial.println("[INFO] Running in Serial-Only Mode without freezing!");
  }

  Serial.println("=========================================\n");
}

void loop() {
  if (wifiConnected) {
    server.handleClient();
  }

  // 1. Read SEN-11574 Pulse Sensor
  rawPulse = analogRead(PULSE_PIN);

  // Auto-track Peak and Valley
  if (rawPulse > peak) peak = rawPulse;
  if (rawPulse < trough) trough = rawPulse;

  // Adapt threshold every 600ms
  if (millis() - lastAdaptTime > 600) {
    lastAdaptTime = millis();
    amplitude = peak - trough;
    dynamicThreshold = trough + (amplitude / 2);

    peak -= (amplitude * 0.2);
    trough += (amplitude * 0.2);
  }

  // Beat Detection
  if (amplitude > 80) {
    if (rawPulse > dynamicThreshold && !pulseDetected && (millis() - lastBeatTime > 300)) {
      pulseDetected = true;
      unsigned long delta = millis() - lastBeatTime;
      lastBeatTime = millis();

      currentBPM = 60000.0 / delta;
      if (currentBPM >= 50.0 && currentBPM <= 190.0) {
        if (smoothedBPM == 0) smoothedBPM = currentBPM;
        else smoothedBPM = (smoothedBPM * 0.65) + (currentBPM * 0.35);
      }
    }

    if (rawPulse < (dynamicThreshold - 30)) {
      pulseDetected = false;
    }
  }

  // Clear BPM if finger is removed
  if (millis() - lastBeatTime > 3500) {
    smoothedBPM = 0;
  }

  // 2. Read Temperature & Print to Serial Monitor every 1 second
  if (millis() - lastTempTime >= 1000) {
    lastTempTime = millis();
    temperatureC = readTemperature();

    // Check health condition
    if (smoothedBPM > 0) {
      bool isTempNormal = (temperatureC >= TEMP_NORMAL_MIN && temperatureC <= TEMP_NORMAL_MAX);
      bool isBpmNormal  = (smoothedBPM >= BPM_NORMAL_MIN && smoothedBPM <= BPM_NORMAL_MAX);

      if (isTempNormal && isBpmNormal) {
        healthStatus = "NORMAL";
      } else {
        healthStatus = "ABNORMAL";
      }
    } else {
      healthStatus = "MEASURING";
    }

    // Save BPM and temperature for the dashboard graphs
    bpmHist[histHead] = smoothedBPM;
    tempHist[histHead] = temperatureC;
    histHead = (histHead + 1) % HIST;
    if (histCount < HIST) histCount++;

    // Always print to Serial Monitor
    Serial.print("Raw Pulse: "); Serial.print(rawPulse);
    Serial.print(" | Threshold: "); Serial.print(dynamicThreshold);
    Serial.print(" | BPM: "); Serial.print(smoothedBPM > 0 ? String(smoothedBPM, 0) : "--");
    Serial.print(" | Temp (SIMULATED): "); Serial.print(temperatureC, 1);
    Serial.print(" °C | Status: "); Serial.println(healthStatus);
  }

  // 3. Play Normal Buzzer Alarms every 4 seconds based on status
  if (millis() - lastBuzzerCheck >= 4000) {
    lastBuzzerCheck = millis();
    if (healthStatus == "NORMAL") {
      playGoodTune(); // 2 short pleasant beeps
    } else if (healthStatus == "ABNORMAL") {
      playBadTune();  // 3 urgent medical alert beeps
    }
  }

  // 4. Send Telegram update every 40 seconds (always)
  if (wifiConnected && (millis() - lastTelegramTime >= TELEGRAM_INTERVAL)) {
    lastTelegramTime = millis();
    sendHealthUpdate();
  }

  delay(15);
}
