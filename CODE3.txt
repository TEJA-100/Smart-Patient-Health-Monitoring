// ================= Pin Assignments =================
#define PULSE_PIN    34   // SEN-11574 Signal pin (ADC1)
#define LM35_AMP_PIN 35   // LM741 Pin 6 Output (ADC1)
#define BUZZER_PIN   25   // Buzzer Positive pin

// ================= Normal Health Thresholds =================
const float TEMP_NORMAL_MIN = 36.0;    // °C
const float TEMP_NORMAL_MAX = 37.8;    // °C
const float BPM_NORMAL_MIN  = 60.0;    // BPM
const float BPM_NORMAL_MAX  = 100.0;   // BPM

// ================= Global Variables =================
WebServer server(80);
bool wifiConnected = false;

int rawPulse = 0;
bool pulseDetected = false;
unsigned long lastBeatTime = 0;
float currentBPM = 0;
float smoothedBPM = 0;
float temperatureC = 0.0;
String healthStatus = "WAITING";

// Dynamic Peak Tracking for BPM
int peak = 2048;
int trough = 2048;
int dynamicThreshold = 2048;
int amplitude = 0;
unsigned long lastAdaptTime = 0;

unsigned long lastTempTime = 0;
unsigned long lastBuzzerCheck = 0;

// ================= NORMAL BUZZER ALARMS =================

// 1. GOOD CONDITION: 2 short, crisp pleasant confirmation beeps
void playGoodTune() {
  tone(BUZZER_PIN, 2000, 70);
  delay(90);
  noTone(BUZZER_PIN);
  delay(60);
  tone(BUZZER_PIN, 2500, 100);
  delay(120);
  noTone(BUZZER_PIN);
}

// 2. BAD CONDITION: Standard urgent medical monitor warning (3 sharp beeps)
void playBadTune() {
  for (int i = 0; i < 3; i++) {
    tone(BUZZER_PIN, 2700, 90);
    delay(110);
    noTone(BUZZER_PIN);
    delay(50);
  }
}

// ================= Web Handlers =================
void handleRoot() {
  server.send(200, "text/html", PAGE_HTML);
}

void handleData() {
  String json = "{";
  json += "\"bpm\":" + String(smoothedBPM, 1) + ",";
  json += "\"temp\":" + String(temperatureC, 2) + ",";
  json += "\"status\":\"" + healthStatus + "\"";
  json += "}";
  server.send(200, "application/json", json);
}

// ================= Read LM35 through LM741 (3x Gain) =================
float readTemperature() {
  long sumMilliVolts = 0;
  for (int i = 0; i < 20; i++) {
    sumMilliVolts += analogReadMilliVolts(LM35_AMP_PIN);
    delay(2);
  }
  float avgMilliVolts = (float)sumMilliVolts / 20.0;
  return avgMilliVolts / 30.0;
}

void setup() {
  // Start Serial Monitor immediately
  Serial.begin(115200);
  delay(1500); // Allow USB-Serial connection to stabilize
  Serial.println("\n=========================================");
  Serial.println("  ESP32 HEALTH MONITOR INITIALIZING...   ");
  Serial.println("=========================================");

  pinMode(BUZZER_PIN, OUTPUT);
  analogReadResolution(12);

  // Power-on test beep
  tone(BUZZER_PIN, 2000, 80);
  delay(100);
  noTone(BUZZER_PIN);

  // Wi-Fi Connection with 10-Second Timeout
  Serial.print("Connecting to Wi-Fi: ");
  Serial.println(ssid);
  WiFi.begin(ssid, password);

  unsigned long wifiStart = millis();
  while (WiFi.status() != WL_CONNECTED && (millis() - wifiStart < 10000)) {
    delay(500);
    Serial.print(".");
  }

  if (WiFi.status() == WL_CONNECTED) {
    wifiConnected = true;
    Serial.println("\n[OK] WiFi Connected!");
    Serial.print("[INFO] Dashboard URL: http://");
    Serial.println(WiFi.localIP());

    server.on("/", handleRoot);
    server.on("/data", handleData);
    server.begin();
  } else {
    Serial.println("\n[WARN] WiFi connection timed out (Wrong SSID/Password or out of range).");
    Serial.println("[INFO] Running in Serial-Only Mode without freezing!");
  }

  Serial.println("=========================================\n");
}

void loop() {
  if (wifiConnected) {
    server.handleClient();
  }

  // 1. Read SEN-11574 Pulse Sensor
  rawPulse = analogRead(PULSE_PIN);

  // Auto-track Peak and Valley
  if (rawPulse > peak) peak = rawPulse;
  if (rawPulse < trough) trough = rawPulse;

  // Adapt threshold every 600ms
  if (millis() - lastAdaptTime > 600) {
    lastAdaptTime = millis();
    amplitude = peak - trough;
    dynamicThreshold = trough + (amplitude / 2);

    peak -= (amplitude * 0.2);
    trough += (amplitude * 0.2);
  }

  // Beat Detection
  if (amplitude > 80) {
    if (rawPulse > dynamicThreshold && !pulseDetected && (millis() - lastBeatTime > 300)) {
      pulseDetected = true;
      unsigned long delta = millis() - lastBeatTime;
      lastBeatTime = millis();

      currentBPM = 60000.0 / delta;
      if (currentBPM >= 50.0 && currentBPM <= 190.0) {
        if (smoothedBPM == 0) smoothedBPM = currentBPM;
        else smoothedBPM = (smoothedBPM * 0.65) + (currentBPM * 0.35);
      }
    }

    if (rawPulse < (dynamicThreshold - 30)) {
      pulseDetected = false;
    }
  }

  // Clear BPM if finger is removed
  if (millis() - lastBeatTime > 3500) {
    smoothedBPM = 0;
  }

  // 2. Read Temperature & Print to Serial Monitor every 1 second
  if (millis() - lastTempTime >= 1000) {
    lastTempTime = millis();
    temperatureC = readTemperature();

    // Check health condition
    if (smoothedBPM > 0) {
      bool isTempNormal = (temperatureC >= TEMP_NORMAL_MIN && temperatureC <= TEMP_NORMAL_MAX);
      bool isBpmNormal  = (smoothedBPM >= BPM_NORMAL_MIN && smoothedBPM <= BPM_NORMAL_MAX);

      if (isTempNormal && isBpmNormal) {
        healthStatus = "GOOD";
      } else {
        healthStatus = "BAD";
      }
    } else {
      healthStatus = "WAITING";
    }

    // Always print to Serial Monitor
    Serial.print("Raw Pulse: "); Serial.print(rawPulse);
    Serial.print(" | Threshold: "); Serial.print(dynamicThreshold);
    Serial.print(" | BPM: "); Serial.print(smoothedBPM > 0 ? String(smoothedBPM, 0) : "--");
    Serial.print(" | Temp: "); Serial.print(temperatureC, 1);
    Serial.print(" °C | Status: "); Serial.println(healthStatus);
  }

  // 3. Play Normal Buzzer Alarms every 4 seconds based on status
  if (millis() - lastBuzzerCheck >= 4000) {
    lastBuzzerCheck = millis();
    if (healthStatus == "GOOD") {
      playGoodTune(); // 2 short pleasant beeps
    } else if (healthStatus == "BAD") {
      playBadTune();  // 3 urgent medical alert beeps
    }
  }

  delay(15);
}
